53:             return NV_FIRMWARE_CHIP_FAMILY_AD10X;

55:         case GPU_ARCHITECTURE_HOPPER:
56:             return NV_FIRMWARE_CHIP_FAMILY_GH100;

58:         case GPU_ARCHITECTURE_BLACKWELL_GB1XX:
59:             if (gpuImpl == GPU_IMPLEMENTATION_GB10B)
60:                 return NV_FIRMWARE_CHIP_FAMILY_GB10Y;
61:             else
62:                 return NV_FIRMWARE_CHIP_FAMILY_GB10X;

64:         case GPU_ARCHITECTURE_BLACKWELL_GB2XX:
65:             if (gpuImpl >= GPU_IMPLEMENTATION_GB20B)
66:                 return NV_FIRMWARE_CHIP_FAMILY_GB20Y;
67:             else
